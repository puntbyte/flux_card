  @override
  State<FluxShowcase> createState() => _FluxShowcaseState();
}

class _FluxShowcaseState extends State<FluxShowcase> {
  ThemeMode _mode = ThemeMode.light;

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      themeMode: _mode,
      theme: ThemeData(
        useMaterial3: true,
        brightness: Brightness.light,
        scaffoldBackgroundColor: const Color(0xFFF8FAFC),
        // Light Slate
        cardColor: Colors.white,
        appBarTheme: const AppBarTheme(backgroundColor: Colors.white, elevation: 0),
      ),
      darkTheme: ThemeData(
        useMaterial3: true,
        brightness: Brightness.dark,
        scaffoldBackgroundColor: const Color(0xFF0F172A), // Deep Navy
        cardColor: const Color(0xFF1E293B), // Slate 800
      ),
      home: DefaultTabController(
        length: 3,
        child: Scaffold(
          appBar: AppBar(
            title: const Text('Flux Card Engine', style: TextStyle(fontWeight: FontWeight.w900)),
            actions: [
              IconButton(
                icon: Icon(_mode == ThemeMode.light ? Icons.dark_mode : Icons.light_mode),
                onPressed: () => setState(
                  () => _mode = _mode == ThemeMode.light ? ThemeMode.dark : ThemeMode.light,
                ),
              ),
            ],
            bottom: const TabBar(
              indicatorSize: TabBarIndicatorSize.label,
              tabs: [
                Tab(text: 'Product'),

            children: [ProductGridPage(), DestinationListPage(), BlogPage()],

        ),
      ),
    );
  }
}

enum ProductCategory { shoes, clothing, electronics, accessories, home, other }

class ProductPricing {
  final double price;
  final double? discount;
  final double? formerPrice;

  const ProductPricing({required this.price, this.discount, this.formerPrice});

  double get currentPrice => (price - (discount ?? 0)).clamp(0, double.infinity);

  bool get hasDiscount => (discount ?? 0) > 0 || formerPrice != null;

  double get savings => formerPrice != null
      ? (formerPrice! - currentPrice).clamp(0, double.infinity)

}

class FluxHeader extends StatelessWidget {
  final Widget? leading;
  final Widget title;
  final Widget? subtitle;
  final List<Widget>? trailing;
  final double? spacing;

  const FluxHeader({
    super.key,
    this.leading,
    required this.title,
    this.subtitle,
    this.trailing,
    this.spacing,
  });

  @override
  Widget build(BuildContext context) {
    final theme = FluxCardTheme.of(context);
    final textTheme = Theme.of(context).textTheme;

    // 1. Resolve styles with non-null fallbacks
    final effectiveTitleStyle = theme.defaultTitleStyle ??
        textTheme.titleMedium ??
        const TextStyle(fontWeight: FontWeight.bold);

    final effectiveSubStyle = theme.defaultSubtitleStyle ??
        textTheme.bodyMedium ??
        const TextStyle();

    final effectiveSpacing = spacing ?? theme.spacing;

    // 2. Extract fields to local variables for Null Safety Promotion
    final Widget? localLeading = leading;
    final Widget? localSubtitle = subtitle;
    final List<Widget>? localTrailing = trailing;

    return Row(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        if (localLeading != null) ...[
          localLeading,
          SizedBox(width: effectiveSpacing),
        ],

        Expanded(
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            mainAxisSize: MainAxisSize.min,
            children: [
              // Use .merge to inherit colors from FluxCard's foregroundColor
              DefaultTextStyle.merge(
                style: effectiveTitleStyle,
                child: title,
              ),

              if (localSubtitle != null) ...[
                SizedBox(height: effectiveSpacing / 2),
                DefaultTextStyle.merge(
                  style: effectiveSubStyle, // Now guaranteed non-null
                  child: localSubtitle,
                ),
              ],
            ],
          ),
        ),

        if (localTrailing != null && localTrailing.isNotEmpty) ...[
          SizedBox(width: effectiveSpacing),
          Wrap(
            spacing: 4,
            alignment: WrapAlignment.end,
            crossAxisAlignment: WrapCrossAlignment.center,
            children: localTrailing,
          ),
        ],
      ],
    );
  }
}

    final isWide = MediaQuery.sizeOf(context).width > effectiveTheme.responsiveBreakpoint;

    final resolvedLayout = (layout.mode == FluxLayoutMode.responsive)
        ? (isWide ? const FluxCardLayout.row() : const FluxCardLayout.column())
        : layout;

    return LayoutBuilder(
      builder: (context, constraints) {
        return SizedBox(
          width: width ?? (fullWidth ? constraints.maxWidth : null),
          height: height ?? (fullHeight ? constraints.maxHeight : null),
          child: Material(
            color: Theme.of(context).cardColor, // Uses the theme's card color
            elevation: effectiveTheme.defaultShadows != null ? 4 : 0,
            shadowColor: Colors.black26,
            shape: RoundedRectangleBorder(borderRadius: effectiveTheme.borderRadius),
            clipBehavior: Clip.antiAlias, // CRITICAL: Clips background/image to radius
            child: InkWell(
              onTap: onTap,
              onLongPress: onLongPress,
              child: DefaultTextStyle.merge(
                style: TextStyle(color: foregroundColor),
                child: IconTheme.merge(
                  data: IconThemeData(color: foregroundColor),
                  child: _buildLayout(context, resolvedLayout, effectiveTheme),
                ),
              ),
            ),
          ),
        );
      },
    );
  }

  Widget _buildLayout(BuildContext context, FluxCardLayout layout, FluxCardThemeData theme) {
    final bg = background ?? Container(color: Theme.of(context).cardColor);

    final inner = SlotResolver.resolve(
      header: header,
      content: content,
      footer: footer,
      padding: theme.padding,
      spacing: theme.spacing,
    );

    if (layout.mode == FluxLayoutMode.stack) {
      return Stack(
        fit: StackFit.passthrough,
        children: [
          Positioned.fill(child: bg),
          if (overlay != null) overlay!,
          inner, // SlotResolver handles its own padding
        ],
      );
    }

    if (layout.mode == FluxLayoutMode.row) {
      return IntrinsicHeight(
        child: Row(
          crossAxisAlignment: CrossAxisAlignment.stretch,
          children: [
            // Use theme-defined flex ratios instead of 2 and 3
            Flexible(flex: theme.flexBackground, child: bg),
            Expanded(flex: theme.flexContent, child: inner),
          ],
        ),
      );
    }

    return Column(
      mainAxisSize: MainAxisSize.min,
      crossAxisAlignment: CrossAxisAlignment.stretch,
      children: [bg, inner],
    );
  }
}

import 'package:flutter/material.dart';

class SlotResolver {
  static Widget resolve({
    Widget? header,
    Widget? content,
    Widget? footer,
    required EdgeInsetsGeometry padding,
    required double spacing,
    Axis direction = Axis.vertical,
  }) {
    final children = <Widget>[];
    if (header != null) children.add(header);
    if (header != null && content != null) children.add(SizedBox(height: spacing));
    if (content != null) children.add(content);
    if (content != null && footer != null) children.add(SizedBox(height: spacing));
    if (footer != null) children.add(footer);

    if (children.isEmpty) return const SizedBox.shrink();

    final layout = direction == Axis.vertical
        ? Column(
            crossAxisAlignment: CrossAxisAlignment.stretch,
            mainAxisSize: MainAxisSize.min,
            children: children,
          )
        : Row(
            crossAxisAlignment: CrossAxisAlignment.start,
            mainAxisSize: MainAxisSize.min,
            children: children,
          );

    return Padding(padding: padding, child: layout);
  }
}
